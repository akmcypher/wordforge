cmake_minimum_required(VERSION 3.25)
project(Wordforge VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets PrintSupport)
pkg_check_modules(ICU REQUIRED IMPORTED_TARGET icu-uc icu-i18n)
pkg_check_modules(TYPOGRAPHY REQUIRED IMPORTED_TARGET harfbuzz freetype2)
pkg_check_modules(XML REQUIRED IMPORTED_TARGET pugixml)
pkg_check_modules(ZIP REQUIRED IMPORTED_TARGET libzip>=1.7)
pkg_check_modules(SKIA REQUIRED IMPORTED_TARGET skia)
pkg_check_modules(HUNSPELL REQUIRED IMPORTED_TARGET hunspell)
if(NOT SKIA_VERSION MATCHES "^143($|[.])")
  message(FATAL_ERROR "Wordforge currently targets Skia m143; validate API changes before upgrading")
endif()
include(CTest)
add_library(wordforge_core core/document/document.cpp core/document/anchors.cpp core/document/images.cpp core/document/tables.cpp core/document/statistics.cpp)
target_include_directories(wordforge_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(wordforge_core PUBLIC PkgConfig::ICU)
add_library(wordforge_layout core/typography/shaper.cpp core/layout/layout.cpp)
target_link_libraries(wordforge_layout PUBLIC wordforge_core PkgConfig::TYPOGRAPHY)
add_library(wordforge_formats core/formats/native.cpp core/formats/docx.cpp core/formats/zip_package.cpp)
target_link_libraries(wordforge_formats PUBLIC wordforge_core PkgConfig::XML PkgConfig::ZIP)
add_library(wordforge_rendering core/rendering/page_renderer.cpp)
target_link_libraries(wordforge_rendering PUBLIC wordforge_layout PkgConfig::SKIA)
if(NOT WIN32)
  find_package(Qt6 6.4 REQUIRED COMPONENTS DBus)
  pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  target_link_libraries(wordforge_rendering PRIVATE PkgConfig::FONTCONFIG)
endif()
add_library(wordforge_desktop apps/desktop/page_view.cpp apps/desktop/page_view.hpp apps/desktop/main_window.cpp apps/desktop/main_window.hpp apps/desktop/appearance.cpp apps/desktop/search_bar.cpp apps/desktop/recovery_file.cpp apps/desktop/bookmarks_dialog.cpp apps/desktop/pictures.cpp apps/desktop/commands.cpp apps/desktop/icons.cpp apps/desktop/shell.cpp apps/desktop/clipboard.cpp apps/desktop/spelling.cpp apps/desktop/self_test.cpp)
set_target_properties(wordforge_desktop PROPERTIES AUTOMOC ON)
# Build identifier shown in About and the Windows file properties; recorded at configure time.
set(WORDFORGE_BUILD_ID "" CACHE STRING "Build identifier (default: current git commit)")
if(NOT WORDFORGE_BUILD_ID)
  execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse --short=12 HEAD
                  OUTPUT_VARIABLE WORDFORGE_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT WORDFORGE_BUILD_ID)
    set(WORDFORGE_BUILD_ID "local")
  endif()
endif()
target_compile_definitions(wordforge_desktop PUBLIC WORDFORGE_VERSION="${PROJECT_VERSION}"
                           WORDFORGE_BUILD_ID="${WORDFORGE_BUILD_ID}")
target_link_libraries(wordforge_desktop PUBLIC wordforge_rendering wordforge_formats Qt6::Widgets Qt6::PrintSupport PRIVATE PkgConfig::HUNSPELL)
if(NOT WIN32)
  target_link_libraries(wordforge_desktop PRIVATE Qt6::DBus)
endif()
add_executable(wordforge apps/desktop/main.cpp)
if(WIN32)
  set_target_properties(wordforge PROPERTIES WIN32_EXECUTABLE TRUE)
  # Icon and version information shown by Explorer and the taskbar.
  enable_language(RC)
  set(WORDFORGE_ICON "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/wordforge.ico")
  set(WORDFORGE_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/wordforge.manifest")
  configure_file(packaging/windows/wordforge.rc.in "${CMAKE_CURRENT_BINARY_DIR}/wordforge.rc" @ONLY)
  # The manifest (UTF-8 code page) is shared with desktop_tests so tests see the same process.
  configure_file(packaging/windows/manifest.rc.in "${CMAKE_CURRENT_BINARY_DIR}/manifest.rc" @ONLY)
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/manifest.rc" APPEND PROPERTY OBJECT_DEPENDS "${WORDFORGE_MANIFEST}")
  target_sources(wordforge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/wordforge.rc" "${CMAKE_CURRENT_BINARY_DIR}/manifest.rc")
endif()
target_link_libraries(wordforge PRIVATE wordforge_desktop)
install(TARGETS wordforge RUNTIME DESTINATION bin)
option(WORDFORGE_BENCHMARKS "Build document performance tools" ON)
if(WORDFORGE_BENCHMARKS)
  add_executable(wordforge_transaction_benchmark benchmarks/transaction_benchmark.cpp)
  target_link_libraries(wordforge_transaction_benchmark PRIVATE wordforge_core)
  if(WIN32)
    target_link_libraries(wordforge_transaction_benchmark PRIVATE psapi)
  endif()
  add_executable(wordforge_desktop_benchmark benchmarks/desktop_benchmark.cpp)
  target_link_libraries(wordforge_desktop_benchmark PRIVATE wordforge_desktop)
  add_executable(wordforge_benchmark benchmarks/document_benchmark.cpp)
  target_link_libraries(wordforge_benchmark PRIVATE wordforge_rendering wordforge_formats)
  if(WIN32)
    target_link_libraries(wordforge_benchmark PRIVATE psapi)
  endif()
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  find_package(GTest CONFIG REQUIRED)
  add_executable(core_tests tests/unit/document_test.cpp tests/unit/table_test.cpp)
  target_link_libraries(core_tests PRIVATE wordforge_core GTest::gtest_main)
  add_test(NAME document COMMAND core_tests)
  add_executable(layout_tests tests/unit/layout_test.cpp)
  target_link_libraries(layout_tests PRIVATE wordforge_layout GTest::gtest_main)
  add_test(NAME layout COMMAND layout_tests)
  add_executable(format_tests tests/unit/format_test.cpp tests/unit/docx_test.cpp)
  target_link_libraries(format_tests PRIVATE wordforge_formats GTest::gtest_main)
  add_test(NAME formats COMMAND format_tests)
  add_executable(render_tests tests/integration/render_test.cpp)
  target_link_libraries(render_tests PRIVATE wordforge_rendering GTest::gtest_main)
  add_test(NAME rendering COMMAND render_tests)
  add_executable(desktop_tests tests/integration/desktop_test.cpp)
  set_target_properties(desktop_tests PROPERTIES AUTOMOC ON)
  target_link_libraries(desktop_tests PRIVATE wordforge_desktop Qt6::Test)
  if(WIN32)
    target_sources(desktop_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/manifest.rc")
  endif()
  add_test(NAME desktop COMMAND desktop_tests)
  set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  # The installation check that packaged builds run, here against the build tree.
  add_test(NAME self_test COMMAND wordforge --self-test "${CMAKE_CURRENT_BINARY_DIR}/self-test-report.txt")
  set_tests_properties(self_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
get_property(wordforge_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS wordforge_targets)
  get_target_property(target_type ${target} TYPE)
  if(target_type STREQUAL "EXECUTABLE" OR target_type STREQUAL "STATIC_LIBRARY")
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()
